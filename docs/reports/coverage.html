func run() error {
        ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
        defer stop()
        secret := os.Getenv("JWT_SECRET")
        if len(secret) < 32 {
                return fmtError("JWT_SECRET must contain at least 32 characters")
        }
        db, e := pgxpool.New(ctx, os.Getenv("DATABASE_URL"))
        if e != nil {
                return e
        }
        defer db.Close()
        if e = db.Ping(ctx); e != nil {
                return e
        }
        opt, e := redis.ParseURL(os.Getenv("REDIS_URL"))
        if e != nil {
                return e
        }
        cache := redis.NewClient(opt)
        defer cache.Close()
        if e = cache.Ping(ctx).Err(); e != nil {
                return e
        }
        store := &storage.Store{DB: db}
        if len(os.Args) == 2 && os.Args[1] == "--reset-demo" {
                if e = store.ResetDemo(ctx); e != nil {
                        return e
                }
                return cache.FlushDB(ctx).Err()
        }
        if e = store.Seed(ctx); e != nil {
                return e
        }
        corp := &corpclient.HTTP{URL: os.Getenv("CORP_MOCK_URL"), Token: os.Getenv("CORP_TOKEN"), Client: &http.Client{Timeout: 2 * time.Second}}
        svc := &service.Service{Repo: store, Corp: corp, Now: time.Now}
        api := &httpapi.API{Store: store, Service: svc, Redis: cache, Secret: secret}
        server := &http.Server{Addr: ":8080", Handler: api, ReadHeaderTimeout: 5 * time.Second, ReadTimeout: 10 * time.Second, WriteTimeout: 15 * time.Second, IdleTimeout: 60 * time.Second}
        done := make(chan error, 1)
        go func() { done <- server.ListenAndServe() }()
        select {
        case e = <-done:
                if e != http.ErrServerClosed {
                        return e
                }
        case <-ctx.Done():
                shutdown, c := context.WithTimeout(context.Background(), 10*time.Second)
                defer c()
                return server.Shutdown(shutdown)
        }
        return nil
}

package auth

import (
        "crypto/hmac"
        "crypto/sha256"
        "encoding/base64"
        "encoding/json"
        "errors"
        "strings"
        "time"
)

type Claims struct {
        Sub string `json:"sub"`
        Exp int64  `json:"exp"`
}

func Issue(id, secret string, now time.Time) string {
        b, _ := json.Marshal(Claims{id, now.Add(12 * time.Hour).Unix()})
        s := base64.RawURLEncoding.EncodeToString([]byte(`{"alg":"HS256","typ":"JWT"}`)) + "." + base64.RawURLEncoding.EncodeToString(b)
        return s + "." + sign(s, secret)
}
func sign(s, key string) string {
        h := hmac.New(sha256.New, []byte(key))
        h.Write([]byte(s))
        return base64.RawURLEncoding.EncodeToString(h.Sum(nil))
}
func Verify(token, secret string, now time.Time) (string, error) {
        p := strings.Split(token, ".")
        if len(p) != 3 || !hmac.Equal([]byte(sign(strings.Join(p[:2], "."), secret)), []byte(p[2])) {
                return "", errors.New("invalid token")
        }
        head, e := base64.RawURLEncoding.DecodeString(p[0])
        var h map[string]string
        if e != nil || json.Unmarshal(head, &h) != nil || h["alg"] != "HS256" {
                return "", errors.New("invalid algorithm")
        }
        b, e := base64.RawURLEncoding.DecodeString(p[1])
        var c Claims
        if e != nil || json.Unmarshal(b, &c) != nil || c.Sub == "" || c.Exp <= now.Unix() {
                return "", errors.New("expired token")
        }
        return c.Sub, nil
}

import (

type Client interface {
        Call(context.Context, string, string, domain.Request, bool) (domain.Request, error)
}
type HTTP struct {
        URL    string
        Token  string
        Client *http.Client
}

func (c *HTTP) Call(ctx context.Context, method, path string, in domain.Request, fail bool) (domain.Request, error) {
        var out domain.Request
        body, e := json.Marshal(in)
        if e != nil {
                return out, e
        }
        var lastErr error
        for attempt := 0; attempt < 3; attempt++ {
                if attempt > 0 {
                        select {
                        case <-ctx.Done():
                                return out, ctx.Err()
                        case <-time.After(time.Duration(attempt) * 100 * time.Millisecond):
                        }
                }
                req, e := http.NewRequestWithContext(ctx, method, c.URL+path, bytes.NewReader(body))
                if e != nil {
                        return out, e
                }
                req.Header.Set("Authorization", "Bearer "+c.Token)
                req.Header.Set("Content-Type", "application/json")
                if fail {
                        req.Header.Set("X-Mock-Fail", "true")
                }
                res, err := c.Client.Do(req)
                if err != nil {
                        lastErr = err
                        continue
                }
                b, err := io.ReadAll(io.LimitReader(res.Body, 1<<20))
                res.Body.Close()
                if err != nil {
                        lastErr = err
                        continue
                }
                if res.StatusCode >= 500 {
                        lastErr = fmt.Errorf("upstream status %d", res.StatusCode)
                        continue
                }
                if res.StatusCode >= 400 {
                        return out, fmt.Errorf("upstream status %d", res.StatusCode)
                }
                e = json.Unmarshal(b, &out)
                return out, e
        }
        return out, lastErr
}

package corpmock

import (
        "aibuddy/backend/internal/domain"
        "crypto/subtle"
        "encoding/json"
        "github.com/jackc/pgx/v5/pgxpool"
        "math/rand"
        "net/http"
        "strings"
        "time"
)

func Handler(db *pgxpool.Pool, token string, rate float64) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                w.Header().Set("Content-Type", "application/json")
                if r.URL.Path == "/healthz" {
                        w.Write([]byte(`{"status":"ok"}`))
                        return
                }
                if subtle.ConstantTimeCompare([]byte(r.Header.Get("Authorization")), []byte("Bearer "+token)) != 1 {
                        http.Error(w, `{"error":{"code":"unauthorized"}}`, http.StatusUnauthorized)
                        return
                }
                select {
                case <-r.Context().Done():
                        return
                case <-time.After(time.Duration(200+rand.Intn(401)) * time.Millisecond):
                }
                if r.Header.Get("X-Mock-Fail") == "true" || rand.Float64() < rate {
                        http.Error(w, `{"error":{"code":"upstream_unavailable"}}`, http.StatusServiceUnavailable)
                        return
                }
                parts := strings.Split(strings.Trim(r.URL.Path, "/"), "/")
                if r.Method == "GET" && r.URL.Path == "/access/systems" {
                        json.NewEncoder(w).Encode([]string{"gitlab", "kafka", "grafana", "storybook", "sentry"})
                        return
                }
                if len(parts) < 2 || (parts[0] != "absence" && parts[0] != "access") || parts[1] != "requests" {
                        http.NotFound(w, r)
                        return
                }
                var req domain.Request
                var data []byte
                switch {
                case len(parts) == 2 && r.Method == "POST":
                        if json.NewDecoder(http.MaxBytesReader(w, r.Body, 65536)).Decode(&req) != nil || req.ID == "" || req.UserID == "" {
                                http.Error(w, "invalid body", http.StatusBadRequest)
                                return
                        }
                        req.Status = "pending"
                        data, _ = json.Marshal(req)
                        _, e := db.Exec(r.Context(), "INSERT INTO corp_requests VALUES($1,$2) ON CONFLICT(id) DO NOTHING", req.ID, data)
                        if e != nil {
                                http.Error(w, "storage unavailable", http.StatusServiceUnavailable)
                                return
                        }
                        e = db.QueryRow(r.Context(), "SELECT data FROM corp_requests WHERE id=$1", req.ID).Scan(&data)
                        if e != nil {
                                http.Error(w, "storage unavailable", http.StatusServiceUnavailable)
                                return
                        }
                case len(parts) == 3 && r.Method == "GET":
                        if db.QueryRow(r.Context(), "SELECT data FROM corp_requests WHERE id=$1", parts[2]).Scan(&data) != nil {
                                http.NotFound(w, r)
                                return
                        }
                case len(parts) == 4 && parts[3] == "decision" && r.Method == "POST":
                        if json.NewDecoder(http.MaxBytesReader(w, r.Body, 65536)).Decode(&req) != nil || (req.Status != "approved" && req.Status != "rejected") {
                                http.Error(w, "invalid decision", http.StatusBadRequest)
                                return
                        }
                        var old []byte
                        if db.QueryRow(r.Context(), "SELECT data FROM corp_requests WHERE id=$1", parts[2]).Scan(&old) != nil {
                                http.NotFound(w, r)
                                return
                        }
                        var existing domain.Request
                        _ = json.Unmarshal(old, &existing)
                        if existing.Status != "pending" && existing.Status != req.Status {
                                http.Error(w, "already decided", http.StatusConflict)
                                return
                        }
                        existing.Status = req.Status
                        existing.DecidedBy = req.DecidedBy
                        existing.DecisionComment = req.DecisionComment
                        existing.UpdatedAt = time.Now()
                        data, _ = json.Marshal(existing)
                        if _, err := db.Exec(r.Context(), "UPDATE corp_requests SET data=$2 WHERE id=$1", parts[2], data); err != nil {
                                http.Error(w, "storage unavailable", http.StatusServiceUnavailable)
                                return
                        }
                default:
                        http.NotFound(w, r)
                        return
                }
                w.Write(data)
        })
}

func write(w http.ResponseWriter, status int, v any) {
        w.Header().Set("Content-Type", "application/json; charset=utf-8")
        w.WriteHeader(status)
        _ = json.NewEncoder(w).Encode(v)
}
func failure(w http.ResponseWriter, e error) {
        var de *domain.Error
        if !errors.As(e, &de) {
                slog.Error("request failed", "error", e)
                de = domain.Err(500, "internal_error", "Внутренняя ошибка")
        }
        write(w, de.Status, map[string]any{"error": de})
}
func decode(r *http.Request, v any) error {
        d := json.NewDecoder(io.LimitReader(r.Body, 65537))
        d.DisallowUnknownFields()
        if e := d.Decode(v); e != nil {
                return domain.Err(400, "validation_error", "Некорректный JSON")
        }
        if d.Decode(new(any)) != io.EOF {
                return domain.Err(400, "validation_error", "Ожидается один JSON объект")
        }
        return nil
}
func (a *API) ServeHTTP(w http.ResponseWriter, r *http.Request) {
        id := service.ID()
        w.Header().Set("X-Request-ID", id)
        start := time.Now()
        uid := ""
        defer func() {
                slog.Info("request", "request_id", id, "user_id", uid, "method", r.Method, "path", r.URL.Path, "duration_ms", time.Since(start).Milliseconds())
        }()
        ctx, cancel := context.WithTimeout(r.Context(), 8*time.Second)
        defer cancel()
        r = r.WithContext(ctx)
        path := strings.TrimPrefix(r.URL.Path, "/api/v1")
        if r.Method == "GET" && (path == "/healthz" || path == "/readyz") {
                if path == "/readyz" {
                        if e := a.Store.DB.Ping(ctx); e != nil {
                                failure(w, domain.Err(503, "unavailable", "База недоступна"))
                                return
                        }
                        if e := a.Redis.Ping(ctx).Err(); e != nil {
                                failure(w, domain.Err(503, "unavailable", "Redis недоступен"))
                                return
                        }
                }
                write(w, 200, map[string]string{"status": "ok"})
                return
        }
        if r.URL.Path == "/api/v1/auth/login" && r.Method == "POST" {
                a.login(w, r)
                return
        }
        token := strings.TrimPrefix(r.Header.Get("Authorization"), "Bearer ")
        var e error
        uid, e = auth.Verify(token, a.Secret, time.Now())
        if e != nil {
                failure(w, domain.Err(401, "unauthorized", "Требуется авторизация"))
                return
        }
        u, e := a.Store.User(ctx, uid)
        if e != nil {
                failure(w, domain.Err(401, "unauthorized", "Неизвестный пользователь"))
                return
        }
        if !strings.HasPrefix(r.URL.Path, "/api/v1/") {
                failure(w, domain.Err(404, "not_found", "Маршрут не найден"))
                return
        }
        // A database advisory lock serializes mutations across API replicas and keeps validation atomic.
        if r.Method != "GET" || strings.HasPrefix(path, "/requests/") {
                conn, err := a.Store.DB.Acquire(ctx)
                if err != nil {
                        failure(w, err)
                        return
                }
                defer conn.Release()
                var locked bool
                if err = conn.QueryRow(ctx, "SELECT pg_try_advisory_lock(814422)").Scan(&locked); err != nil {
                        failure(w, err)
                        return
                }
                if !locked {
                        failure(w, domain.Err(409, "conflict", "Another mutation is in progress; retry with the same key"))
                        return
                }
                defer func() {
                        unlock, c := context.WithTimeout(context.Background(), 2*time.Second)
                        defer c()
                        _, _ = conn.Exec(unlock, "SELECT pg_advisory_unlock(814422)")
                }()
        }
        var out any

                if e == nil {
                        data, _ := json.Marshal(b)
                        _, e = a.Store.DB.Exec(ctx, "INSERT INTO events(user_id,data) VALUES($1,$2)", u.ID, data)
                }
                status = 201
                out = map[string]bool{"saved": true}
        case path == "/chat/messages" && r.Method == "POST":
                var b struct {
                        ConversationID string         `json:"conversation_id"`
                        Role           string         `json:"role"`
                        Content        string         `json:"content"`
                        Meta           map[string]any `json:"meta"`
                }
                e = decode(r, &b)
                if e == nil && (b.ConversationID == "" || len(b.Content) > 16000 || (b.Role != "user" && b.Role != "assistant" && b.Role != "tool")) {
                        e = domain.Err(400, "validation_error", "Некорректное сообщение")
                }
                var messageID int64
                if e == nil {
                        data, _ := json.Marshal(b)
                        e = a.Store.DB.QueryRow(ctx, "INSERT INTO chat_messages(user_id,conversation_id,data) VALUES($1,$2,$3) RETURNING id", u.ID, b.ConversationID, data).Scan(&messageID)
                }
                out = map[string]any{"id": messageID}
                status = 201
        case path == "/chat/history" && r.Method == "GET":
                out, e = a.history(r, u)
        default:
                e = domain.Err(404, "not_found", "Маршрут не найден")
        }
        if e != nil {
                failure(w, e)
                return
        }
        write(w, status, out)
}
func (a *API) login(w http.ResponseWriter, r *http.Request) {
        var b struct {
                Login    string `json:"login"`
                Password string `json:"password"`
        }
        if e := decode(r, &b); e != nil {
                failure(w, e)
                return
        }
        key := fmt.Sprintf("login:%x", sha256.Sum256([]byte(b.Login)))
        n, e := a.Redis.Eval(r.Context(), `local n=redis.call('INCR',KEYS[1]); if n==1 then redis.call('EXPIRE',KEYS[1],60) end; return n`, []string{key}).Int()
        if e != nil {
                failure(w, domain.Err(503, "unavailable", "Redis недоступен"))
                return
        }
        if n > 10 {
                failure(w, domain.Err(429, "rate_limited", "Слишком много попыток"))
                return
        }
        var id string
        e = a.Store.DB.QueryRow(r.Context(), "SELECT id FROM users WHERE login=$1", b.Login).Scan(&id)
        u, err := a.Store.User(r.Context(), id)
        if e != nil || err != nil || bcrypt.CompareHashAndPassword([]byte(u.PasswordHash), []byte(b.Password)) != nil {
                failure(w, domain.Err(401, "unauthorized", "Неверные учётные данные"))
                return
        }
        write(w, 200, map[string]any{"access_token": auth.Issue(u.ID, a.Secret, time.Now()), "user": u})
}
func (a *API) create(w http.ResponseWriter, r *http.Request, u domain.User, kind string) {
        key := r.Header.Get("Idempotency-Key")
        if key == "" || len(key) > 200 {
                failure(w, domain.Err(400, "validation_error", "Нужен Idempotency-Key до 200 символов"))
                return
        }
        var p map[string]string
        if e := decode(r, &p); e != nil {
                failure(w, e)
                return
        }
        body, _ := json.Marshal(p)
        hash := fmt.Sprintf("%x", sha256.Sum256(append([]byte(kind), body...)))
        redisKey := "idem:" + u.ID + ":" + key
        var cached struct {
                Hash     string          `json:"hash"`
                Response json.RawMessage `json:"response"`
        }
        if a.cached(r.Context(), redisKey, &cached) && cached.Hash != "" {
                if cached.Hash != hash {
                        failure(w, domain.Err(409, "conflict", "Idempotency key has a different payload"))
                        return
                }
                write(w, 200, cached.Response)
                return
        }
        var oldhash string
        var response []byte
        e := a.Store.DB.QueryRow(r.Context(), "SELECT body_hash,response FROM idempotency WHERE user_id=$1 AND key=$2 AND created_at>now()-interval '24 hours'", u.ID, key).Scan(&oldhash, &response)
        if e == nil {
                if oldhash != hash {
                        failure(w, domain.Err(409, "conflict", "Ключ уже использован с другим телом"))
                        return
                }
                var v any
                _ = json.Unmarshal(response, &v)
                write(w, 200, v)
                return
        }
        if !errors.Is(e, pgx.ErrNoRows) {
                failure(w, e)
                return
        }
        svc := *a.Service
        svc.RequestID = fmt.Sprintf("%x", sha256.Sum256([]byte(u.ID+":"+key+":"+hash)))
        svc.Repo = &idemRepository{Store: a.Store, key: key, hash: hash}
        req, e := svc.Create(r.Context(), u, kind, p, r.Header.Get("X-Mock-Fail") == "true")
        if e != nil {
                failure(w, e)
                return
        }
        response, _ = json.Marshal(req)
        cached.Hash = hash
        cached.Response = response
        encoded, _ := json.Marshal(cached)
        _ = a.Redis.Set(r.Context(), redisKey, encoded, 24*time.Hour).Err()
        write(w, 201, req)
}

        }

func (a *API) search(r *http.Request, u domain.User) (any, error) {
        q := r.URL.Query().Get("query")
        if strings.TrimSpace(q) == "" {
                return nil, domain.Err(400, "validation_error", "Нужен query")
        }
        rows, e := a.Store.DB.Query(r.Context(), `SELECT id,title,ts_headline('russian',body,plainto_tsquery('russian',$1)),team_id,ts_rank(tsv,plainto_tsquery('russian',$1)) FROM knowledge_articles WHERE (team_id IS NULL OR team_id=$2) AND tsv @@ plainto_tsquery('russian',$1) ORDER BY 5 DESC,id LIMIT $3`, q, u.TeamID, limit(r, 5))
        if e != nil {
                return nil, e
        }
        defer rows.Close()
        out := []map[string]any{}
        for rows.Next() {
                var id, title, snippet string
                var team *string
                var rank float32
                if e = rows.Scan(&id, &title, &snippet, &team, &rank); e != nil {
                        return nil, e
                }
                out = append(out, map[string]any{"id": id, "title": title, "snippet": snippet, "team_id": team, "rank": rank, "is_test_data": true})
        }
        return out, rows.Err()
}
func (a *API) history(r *http.Request, u domain.User) (any, error) {

                }

        }
        return out, rows.Err()
}

func (s *idemRepository) SaveRequest(ctx context.Context, r domain.Request) error {
        tx, e := s.DB.Begin(ctx)
        if e != nil {
                return e
        }
        defer tx.Rollback(ctx)
        b, e := json.Marshal(r)
        if e != nil {
                return e
        }
        if _, e = tx.Exec(ctx, "SELECT save_requests($1)", b); e != nil {
                return e
        }
        if r.Status != "failed" {
                if _, e = tx.Exec(ctx, "INSERT INTO idempotency(user_id,key,body_hash,response) VALUES($1,$2,$3,$4) ON CONFLICT(user_id,key) DO UPDATE SET body_hash=excluded.body_hash,response=excluded.response,created_at=now()", r.UserID, s.key, s.hash, b); e != nil {
                        return e
                }
        }
        return tx.Commit(ctx)
}

func Can(u domain.User, action string, r domain.Resource) bool {
        switch action {
        case "profile", "create", "chat", "events":
                return u.ID != "" && r.UserID == u.ID
        case "team", "knowledge":
                return u.ID != "" && (r.TeamID == "" || u.TeamID == r.TeamID)
        case "plan":
                return u.ID == r.UserID || u.Role == "buddy" && r.BuddyID == u.ID || u.Role == "lead" && u.TeamID == r.TeamID
        case "task":
                return u.Role == "intern" && u.ID == r.UserID
        case "request":
                return u.ID == r.UserID || u.Role == "lead" && u.TeamID == r.TeamID
        case "approve":
                return u.Role == "lead" && u.TeamID == r.TeamID && u.ID != r.UserID
        case "pending":
                return u.Role == "lead" && u.TeamID == r.TeamID
        }
        return false
}

func ValidateLeave(p map[string]string, now time.Time, existing []domain.Request) error {
        from, e := time.Parse("2006-01-02", p["date_from"])
        to, e2 := time.Parse("2006-01-02", p["date_to"])
        today := time.Date(now.Year(), now.Month(), now.Day(), 0, 0, 0, 0, time.UTC)
        if e != nil || e2 != nil || !from.After(today) || to.Before(from) || to.Sub(from) > 13*24*time.Hour || (p["reason_type"] != "day_off" && p["reason_type"] != "vacation") {
                return domain.Err(400, "validation_error", "Укажите будущие даты, период до 14 дней и корректный reason_type")
        }
        for _, r := range existing {
                if r.Type == "leave" && (r.Status == "pending" || r.Status == "approved") && p["date_from"] <= r.Payload["date_to"] && p["date_to"] >= r.Payload["date_from"] {
                        return domain.Err(409, "conflict", "Период пересекается с активной заявкой")
                }
        }
        return nil
}

                if e != nil {
                        return nil, e

        }

}
func (s *Service) Overview(ctx context.Context, u domain.User) ([]map[string]any, error) {
        var team domain.Team
        if e := s.Repo.Get(ctx, "teams", u.TeamID, &team); e != nil {
                return nil, e
        }
        all, e := s.Repo.Requests(ctx)
        if e != nil {
                return nil, e
        }
        out := []map[string]any{}
        for _, t := range team.Tools {
                status, id := "missing", ""
                for _, r := range all {
                        if r.UserID != u.ID || r.Type != "access" || r.Payload["system_code"] != t.Code {
                                continue
                        }
                        if r.Status == "approved" {
                                status, id = "granted", r.ID
                        } else if r.Status == "pending" && status != "granted" {
                                status, id = "requested", r.ID
                        }
                }
                out = append(out, map[string]any{"system_code": t.Code, "title": t.Title, "purpose": t.Purpose, "required": t.Required, "status": status, "request_id": id})
        }
        return out, nil
}
func (s *Service) Create(ctx context.Context, u domain.User, kind string, p map[string]string, fail bool) (domain.Request, error) {
        var r domain.Request
        if len(p["comment"]) > 2000 || len(p["reason"]) > 2000 {
                return r, domain.Err(400, "validation_error", "Слишком длинный текст")
        }
        if kind == "leave" {
                all, e := s.Visible(ctx, u)
                if e != nil {
                        return r, e
                }
                own := []domain.Request{}
                for _, x := range all {
                        if x.UserID == u.ID {
                                own = append(own, x)
                        }
                }
                if e = ValidateLeave(p, s.Now(), own); e != nil {
                        return r, e
                }
        } else {
                items, e := s.Overview(ctx, u)
                if e != nil {
                        return r, e
                }
                found := false
                for _, t := range items {
                        if t["system_code"] == p["system_code"] {
                                found = true
                                if t["status"] != "missing" {
                                        return r, domain.Err(409, "conflict", "Доступ уже выдан или запрошен")
                                }
                        }
                }
                if !found || strings.TrimSpace(p["reason"]) == "" {
                        return r, domain.Err(400, "validation_error", "Нужна система вашей команды и причина")
                }
        }
        r = domain.Request{ID: ID(), UserID: u.ID, Type: kind, Payload: p, Status: "pending", CreatedAt: s.Now(), UpdatedAt: s.Now()}
        if s.RequestID != "" {
                r.ID = s.RequestID
        }
        external, e := s.Corp.Call(ctx, "POST", corpPath(kind), r, fail)
        if e != nil {
                r.Status = "failed"
        } else {
                r.ExternalID = external.ID
        }
        if err := s.Repo.SaveRequest(ctx, r); err != nil {
                return r, err
        }
        if e != nil {
                return r, &domain.Error{Status: 503, Code: "upstream_unavailable", Message: "Корпоративный сервис недоступен", Details: map[string]any{"retryable": true, "request_id": r.ID}}
        }
        return r, nil
}
func corpPath(kind string) string {
        if kind == "leave" {
                return "/absence/requests"
        }
        return "/access/requests"
}
func (s *Service) Request(ctx context.Context, u domain.User, id string) (domain.Request, error) {
        var r domain.Request
        if e := s.Repo.Get(ctx, "requests", id, &r); e != nil {
                return r, domain.Err(404, "not_found", "Заявка не найдена")
        }
        owner, e := s.Repo.User(ctx, r.UserID)
        if e != nil {
                return r, e
        }
        if !rbac.Can(u, "request", Resource(owner)) {
                return domain.Request{}, domain.Err(403, "forbidden", "Нет доступа к заявке")
        }
        return r, nil
}
func (s *Service) Decision(ctx context.Context, u domain.User, id, status, comment string, fail bool) (domain.Request, error) {
        r, e := s.Request(ctx, u, id)
        if e != nil {
                return r, e
        }
        owner, e := s.Repo.User(ctx, r.UserID)
        if e != nil {
                return r, e
        }
        if !rbac.Can(u, "approve", Resource(owner)) {
                return r, domain.Err(403, "forbidden", "Только руководитель команды, кроме своей заявки")
        }
        if r.Status != "pending" {
                return r, domain.Err(409, "conflict", "Заявка уже рассмотрена")
        }
        r.Status = status
        r.DecidedBy = u.ID
        r.DecisionComment = comment
        r.UpdatedAt = s.Now()
        if _, e = s.Corp.Call(ctx, "POST", corpPath(r.Type)+"/"+r.ExternalID+"/decision", r, fail); e != nil {
                return r, &domain.Error{Status: 503, Code: "upstream_unavailable", Message: "Не удалось согласовать", Details: map[string]bool{"retryable": true}}
        }
        if e = s.Repo.SaveRequest(ctx, r); e != nil {
                return r, e
        }
        return r, s.CloseAccess(ctx, owner)
}
func (s *Service) CloseAccess(ctx context.Context, u domain.User) error {
        items, e := s.Overview(ctx, u)
        if e != nil {
                return e
        }
        for _, t := range items {
                if t["required"] == true && t["status"] != "granted" {
                        return nil
                }
        }
        p, e := s.Repo.Plan(ctx, u.ID)
        if e != nil {
                return e
        }
        for i := range p {
                for j := range p[i].Tasks {
                        if p[i].Tasks[j].Code == "get_access" {
                                p[i].Tasks[j].Status = "done"
                        }
                }
        }
        return s.Repo.SavePlan(ctx, u.ID, p)
}

        target, e := s.Repo.User(ctx, id)
        if e != nil {

        }
        if !rbac.Can(u, "plan", Resource(target)) {
                return nil, domain.Err(403, "forbidden", "Нет доступа к плану")
        }
        return s.Repo.Plan(ctx, id)
}
func (s *Service) Task(ctx context.Context, u domain.User, id, status, comment string) error {
        if !rbac.Can(u, "task", Resource(u)) {
                return domain.Err(403, "forbidden", "Изменять задачи может только стажёр")
        }
        if status != "todo" && status != "in_progress" && status != "done" && status != "blocked" {
                return domain.Err(400, "validation_error", "Неизвестный статус")
        }
        p, e := s.Repo.Plan(ctx, u.ID)
        if e != nil {
                return e
        }
        for i := range p {
                for j := range p[i].Tasks {
                        t := &p[i].Tasks[j]
                        if t.ID == id {
                                if t.Code == "get_access" {
                                        return domain.Err(409, "conflict", "Статус доступов изменяется автоматически")
                                }
                                t.Status = status
                                t.Comment = comment
                                return s.Repo.SavePlan(ctx, u.ID, p)
                        }
                }
        }
        return domain.Err(404, "not_found", "Задача не найдена")
}
func Progress(p []domain.Stage) map[string]any {
        total, done := 0, 0
        stages := []map[string]any{}
        current := ""
        next := []domain.Task{}
        for _, stage := range p {
                n, d := 0, 0
                pending := []domain.Task{}
                for _, t := range stage.Tasks {
                        if t.Required {
                                n++
                                if t.Status == "done" {
                                        d++
                                } else {
                                        pending = append(pending, t)
                                }
                        }
                }
                if current == "" && len(pending) > 0 {
                        current = stage.ID
                        next = pending
                        if len(next) > 3 {
                                next = next[:3]
                        }
                }
                pct := 100
                if n > 0 {
                        pct = d * 100 / n
                }
                stages = append(stages, map[string]any{"stage_id": stage.ID, "percent": pct})
                total += n
                done += d
        }
        pct := 100
        if total > 0 {
                pct = done * 100 / total
        }
        return map[string]any{"percent": pct, "stages": stages, "current_stage": current, "next_steps": next}
}

// Sync refreshes a visible request from the corporate system.
func (s *Service) Sync(ctx context.Context, u domain.User, id string) (domain.Request, error) {
        r, e := s.Request(ctx, u, id)
        if e != nil || r.ExternalID == "" || r.Status == "failed" {
                return r, e
        }
        remote, e := s.Corp.Call(ctx, "GET", corpPath(r.Type)+"/"+r.ExternalID, r, false)
        if e != nil {
                return r, &domain.Error{Status: 503, Code: "upstream_unavailable", Message: "Не удалось обновить статус", Details: map[string]bool{"retryable": true}}
        }
        if remote.Status != "pending" && remote.Status != "approved" && remote.Status != "rejected" {
                return r, domain.Err(503, "upstream_unavailable", "Некорректный статус сервиса")
        }
        if remote.Status != r.Status {
                r.Status = remote.Status
                r.UpdatedAt = s.Now()
                if e = s.Repo.SaveRequest(ctx, r); e != nil {
                        return r, e
                }
                owner, e := s.Repo.User(ctx, r.UserID)
                if e != nil {
                        return r, e
                }
                if r.Type == "access" {
                        e = s.CloseAccess(ctx, owner)
                }
                return r, e
        }
        return r, nil
}

// ResetDemo resets only the active synthetic dataset; migration snapshots remain intact.
func (s *Store) ResetDemo(ctx context.Context) error {
        tx, e := s.DB.Begin(ctx)
        if e != nil {
                return e
        }
        defer tx.Rollback(ctx)
        if _, e = tx.Exec(ctx, "SELECT pg_advisory_xact_lock(814422)"); e != nil {
                return e
        }
        if _, e = tx.Exec(ctx, "TRUNCATE idempotency,corp_requests,chat_messages,events,requests,user_tasks,onboarding_tasks,onboarding_stages,knowledge_articles,team_tools,users,teams RESTART IDENTITY"); e != nil {
                return e
        }
        if e = s.seedTx(ctx, tx); e != nil {
                return e
        }
        return tx.Commit(ctx)
}

package storage

import (
        "aibuddy/backend/internal/domain"
        "context"
        "encoding/json"
        "github.com/jackc/pgx/v5/pgxpool"
)

type Store struct{ DB *pgxpool.Pool }

func (s *Store) User(ctx context.Context, id string) (domain.User, error) {
        var u domain.User
        var b []byte
        err := s.DB.QueryRow(ctx, "SELECT a.data,u.password_hash FROM users u JOIN users_api a USING(id) WHERE id=$1", id).Scan(&b, &u.PasswordHash)
        if err == nil {
                err = json.Unmarshal(b, &u)
        }
        return u, err
}
func (s *Store) Get(ctx context.Context, table, id string, out any) error {
        var b []byte
        err := s.DB.QueryRow(ctx, "SELECT data FROM "+table+"_api WHERE id=$1", id).Scan(&b)
        if err != nil {
                return err
        }
        return json.Unmarshal(b, out)
}
func (s *Store) Requests(ctx context.Context) ([]domain.Request, error) {
        rows, e := s.DB.Query(ctx, "SELECT data FROM requests_api ORDER BY data->>'created_at',id")
        if e != nil {
                return nil, e
        }
        defer rows.Close()
        a := []domain.Request{}
        for rows.Next() {
                var b []byte
                var r domain.Request
                if e = rows.Scan(&b); e != nil {
                        return nil, e
                }
                if e = json.Unmarshal(b, &r); e != nil {
                        return nil, e
                }
                a = append(a, r)
        }
        return a, rows.Err()
}
func (s *Store) SaveRequest(ctx context.Context, r domain.Request) error {
        b, e := json.Marshal(r)
        if e != nil {
                return e
        }
        _, e = s.DB.Exec(ctx, "SELECT save_requests($1)", b)
        return e
}
func (s *Store) Plan(ctx context.Context, id string) ([]domain.Stage, error) {
        var b []byte
        a := []domain.Stage{}
        e := s.DB.QueryRow(ctx, "SELECT data FROM user_plans_api WHERE user_id=$1", id).Scan(&b)
        if e == nil {
                e = json.Unmarshal(b, &a)
        }
        return a, e
}
func (s *Store) SavePlan(ctx context.Context, id string, p []domain.Stage) error {
        b, e := json.Marshal(p)
        if e != nil {
                return e
        }
        _, e = s.DB.Exec(ctx, "SELECT save_plan($1,$2)", id, b)
        return e
}
